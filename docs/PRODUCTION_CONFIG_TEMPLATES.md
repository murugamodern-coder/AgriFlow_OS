## 3. Nginx Configuration

Location: `/etc/nginx/conf.d/agriflow.conf`

```nginx
server {
    listen 80;
    server_name agriflow.yourcompany.com;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name agriflow.yourcompany.com;
    
    ssl_certificate /etc/letsencrypt/live/agriflow.yourcompany.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/agriflow.yourcompany.com/privkey.pem;
    
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;
    
    client_max_body_size 100M;
    
    root /home/agriflow/frappe-bench/sites;
    
    location /assets {
        try_files $uri =404;
    }
    
    location ~ ^/protected/(.*) {
        internal;
        try_files /agriflow.yourcompany.com/$1 =404;
    }
    
    location /socket.io {
        proxy_pass http://127.0.0.1:9000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
    
    location / {
        try_files /agriflow.yourcompany.com$uri @webserver;
    }
    
    location @webserver {
        proxy_pass http://127.0.0.1:8000;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Frappe-Site-Name agriflow.yourcompany.com;
        proxy_read_timeout 120;
        proxy_redirect off;
    }
    
    # Security headers
    add_header X-Frame-Options "SAMEORIGIN";
    add_header X-Content-Type-Options "nosniff";
    add_header Referrer-Policy "no-referrer-when-downgrade";
}
```
